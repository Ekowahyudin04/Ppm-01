class MyHomePage extends StatefulWidget {

  State<MyHomePage> createState() => _MyHomePageState();
}

class _MyHomePageState extends State<MyHomePage> {

  // Fungsi untuk menambah counter
  void _incrementCounter() {
    setState(() {
      _counter++;
    });
  }

  // Fungsi untuk mengurang counter dengan pengecekan tidak boleh < 0
  void _decrementCounter() {
    if (_counter > 0) {
      setState(() {
        _counter--;
      });
    } else {
      // Menampilkan SnackBar jika ditekan saat angka 0
      ScaffoldMessenger.of(context).clearSnackBars();
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Counter tidak boleh kurang dari 0!'),
          backgroundColor: Colors.redAccent,
          duration: Duration(seconds: 2),
        ),
      );
    }
  }

  // Fungsi untuk mereset counter ke 0
  void _resetCounter() {
    setState(() {
      _counter = 0;
    });
  }
